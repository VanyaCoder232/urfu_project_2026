// Вспомогательная функция, чтобы не дублировать код вызова текста
function showDialogue(runtime, position, text) {
    const isTop = position === "top";
    runtime.objects[isTop ? "DiagPanelTop" : "DiagPanelBottom"].getFirstInstance().isVisible = true;
    runtime.objects[isTop ? "DiagTextTop" : "DiagTextBottom"].getFirstInstance().text = text;
}
